# Ties Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Ties, a native macOS app that imports Apple Contacts, researches each person on the public web without an LLM, extracts a structured profile with a user-chosen AI provider, and answers "who do I know that can help with X?".

**Architecture:** A UI-free Swift package `TiesCore` (models, GRDB store, research probes, scoring, AI providers, search) that is fully unit-tested with `swift test`, plus a SwiftUI app target `Ties` generated by xcodegen that contains the setup wizard, the three-column main window, the WebKit search backend, and the Apple on-device provider. Everything is local: one SQLite file, keys in Keychain.

**Tech Stack:** Swift 6 (strict concurrency), SwiftUI, AppKit, Contacts, WebKit, NaturalLanguage, FoundationModels (macOS 26 only, weak-linked), GRDB.swift 7.11, SwiftSoup 2.13, PhoneNumberKit 4.3, Swift Testing, xcodegen 2.46.

**Spec:** `docs/superpowers/specs/2026-09-16-ties-design.md`

## Global Constraints

- Minimum OS: macOS 15. Anything from macOS 26 is behind `if #available(macOS 26, *)`.
- Swift language mode 6 with `SWIFT_STRICT_CONCURRENCY = complete`. Every model type is `Sendable`.
- Exactly three third-party dependencies: GRDB.swift 7.11, SwiftSoup 2.13, PhoneNumberKit 4.3. No others.
- App Sandbox OFF, Hardened Runtime ON, ad-hoc code signature (`CODE_SIGN_IDENTITY = -`).
- Entitlements: `com.apple.security.personal-information.addressbook`, `com.apple.security.network.client`.
- Info.plist: `NSContactsUsageDescription = "Ties reads your contacts to build profiles you can search."`
- Database file: `~/Library/Application Support/Ties/ties.sqlite`. API keys in Keychain service `Ties` with account `<providerId>`.
- LinkedIn URLs are never fetched. Only search snippets are stored for them.
- UI copy: no sentence longer than one line, no subtitles, SF Symbols carry meaning. Animations use `.snappy`.
- Tests use Swift Testing (`import Testing`, `@Test`, `#expect`). Run with `cd TiesCore && swift test`.
- Commit after every task with a conventional-commit message. Never commit `.build/`, `DerivedData/`, `*.xcodeproj` (generated).

public struct Person: Identifiable {
    public var id: String              // UUID string
    public var cnIdentifier: String?
    public var givenName: String
    public var familyName: String
    public var displayName: String
    public var organization: String?
    public var jobTitle: String?
    public var thumbnail: Data?
    public var source: Source          // enum Source: String { case contacts, manual }
    public var createdAt: Date
    public var updatedAt: Date
    public init(id: String = UUID().uuidString, cnIdentifier: String? = nil, givenName: String, familyName: String,
                displayName: String? = nil, organization: String? = nil, jobTitle: String? = nil,
                thumbnail: Data? = nil, source: Source = .contacts, createdAt: Date = .now, updatedAt: Date = .now)
    // displayName defaults to "\(givenName) \(familyName)".trimmed, or organization if both names empty
}
public struct Channel { public enum Kind: String, Codable, Sendable { case phone, email, url }
    public var personId: String; public var kind: Kind; public var label: String?; public var value: String; public var normalized: String }
public struct Candidate: Identifiable { public enum Status: String, Codable, Sendable { case auto, pending, accepted, rejected }
    public var id: String; public var personId: String; public var score: Double; public var status: Status
    public var displayName: String?; public var headline: String?; public var company: String?; public var location: String?
    public var avatarURL: String?; public var primaryURL: String }
public struct Evidence: Identifiable { public enum Kind: String, Codable, Sendable { case emailHash, phone, company, location, name, username, avatar, conflict }
    public var id: String; public var candidateId: String; public var kind: Kind; public var weight: Double; public var detail: String; public var sourceURL: String? }
public struct SourcePage: Identifiable { public enum Kind: String, Codable, Sendable { case gravatar, github, serp, page, username }
    public var id: String; public var candidateId: String; public var url: String; public var title: String?; public var snippet: String?
    public var bodyText: String?; public var fetchedAt: Date; public var kind: Kind }
public struct Fact: Codable, Hashable, Sendable { public var text: String; public var sources: [String]; public init(text: String, sources: [String] = []) }
public struct ProfileFacts: Codable, Hashable, Sendable {
    public var occupation: String?; public var summary: String?
    public var companies: [Fact]; public var achievements: [Fact]; public var certificates: [Fact]; public var experience: [Fact]
    public var canHelpWith: [String]
    public init(occupation: String? = nil, summary: String? = nil, companies: [Fact] = [], achievements: [Fact] = [],
                certificates: [Fact] = [], experience: [Fact] = [], canHelpWith: [String] = [])
    public static let empty = ProfileFacts()
    public var isEmpty: Bool
    public var searchableText: String   // occupation, summary, every Fact.text, every tag, joined by "\n"
}
public struct Profile { public var personId: String; public var facts: ProfileFacts; public var confidence: Double
    public var providerId: String; public var model: String?; public var extractedAt: Date; public var embedding: [Float]? }
public struct Note { public var personId: String; public var body: String; public var updatedAt: Date }
public struct Job: Identifiable { public enum Kind: String, Codable, Sendable { case scan, extract }
    public enum State: String, Codable, Sendable { case queued, running, done, failed, skipped }
    public var id: String; public var kind: Kind; public var personId: String; public var state: State; public var error: String?; public var updatedAt: Date }

public var searchableText: String {

public final class Store: Sendable {
    public static func open(at url: URL) throws -> Store      // creates parent dir, DatabasePool, runs migrations
    public static func inMemory() throws -> Store             // DatabaseQueue(":memory:") for tests
    public static var defaultURL: URL                         // ~/Library/Application Support/Ties/ties.sqlite
    public let writer: any DatabaseWriter
    public func deleteEverything() throws                     // drops all rows (not the file)
    public func databaseSizeBytes() -> Int64
    public func exportJSON() throws -> Data                   // [{person, channels, profile, note}] pretty-printed
}
// Store+People
extension Store {
    public func upsertPeople(_ people: [Person], channels: [Channel]) throws   // by cnIdentifier when present, else id; replaces that person's channels
    public func insertManualPerson(_ person: Person, channels: [Channel]) throws
    public func updatePerson(_ person: Person, channels: [Channel]) throws
    public func deletePerson(id: String) throws                                 // cascades
    public func allPeople() throws -> [Person]                                  // sorted by displayName, localized case-insensitive
    public func person(id: String) throws -> Person?
    public func channels(personId: String) throws -> [Channel]
    public func channelsByPerson(ids: [String]) throws -> [String: [Channel]]
    public func people(matchingNameOrDigits query: String) throws -> [Person]   // name contains (diacritic-insensitive) OR any phone.normalized contains digits-only query
}
// Store+Candidates
extension Store {
    public func replaceCandidates(personId: String, candidates: [Candidate], evidence: [Evidence], pages: [SourcePage]) throws // deletes old ones for personId first
    public func candidates(personId: String) throws -> [Candidate]              // sorted by score desc
    public func evidence(candidateId: String) throws -> [Evidence]
    public func pages(candidateId: String) throws -> [SourcePage]
    public func pagesForAccepted(personId: String) throws -> [SourcePage]       // pages of candidates with status auto/accepted
    public func setCandidateStatus(id: String, status: Candidate.Status) throws // accepting one sets all other candidates of that person to rejected
    public func bestCandidate(personId: String) throws -> Candidate?            // status accepted, else auto, else highest pending
    public func bestCandidatesByPerson() throws -> [String: Candidate]
}
// Store+Profiles
extension Store {
    public func upsertProfile(_ profile: Profile) throws     // also rewrites profile_fts row
    public func profile(personId: String) throws -> Profile?
    public func profilesByPerson() throws -> [String: Profile]
    public func upsertNote(_ note: Note) throws              // also rewrites profile_fts row
    public func note(personId: String) throws -> Note?
    public func ftsSearch(_ query: String, limit: Int = 50) throws -> [(personId: String, rank: Double)]  // FTS5 MATCH with prefix pattern, ordered by bm25
    public func allEmbeddings() throws -> [(personId: String, vector: [Float])]
}
// Store+Jobs
extension Store {
    public func enqueue(kind: Job.Kind, personIds: [String]) throws   // one queued job per person, replacing any existing job of that kind for that person
    public func setJob(kind: Job.Kind, personId: String, state: Job.State, error: String? = nil) throws
    public func jobs(kind: Job.Kind) throws -> [Job]
    public func counts(kind: Job.Kind) throws -> [Job.State: Int]
}

    public var emailHash = 8.0, phone = 6.0, company = 3.0, location = 1.5, avatar = 2.0, username = 1.5, conflict = -3.0
    public var autoThreshold = 6.0, pendingThreshold = 2.0
    public static let `default` = ScoringWeights()
}

    public var candidate: Candidate; public var evidence: [Evidence]; public var pages: [SourcePage]
}
public enum CandidateGrouper {
    /// Groups findings into identities: same canonical url → same group; groups are merged when they share a username,
    /// an avatarURL, or one finding's linkedURLs contains the other's url.
    public static func group(_ findings: [ProbeFinding]) -> [[ProbeFinding]]
}
public enum CandidateScorer {
    public static func score(groups: [[ProbeFinding]], input: ProbeInput, weights: ScoringWeights = .default) -> [ScoredCandidate]
    // Per group: name gate — at least one finding must have displayName with similarity ≥ NameMatcher.gate, OR any emailHash evidence,
    //   OR a bodyText containing the name. Groups failing the gate are dropped.
    // Evidence: union of finding evidence; keep the max weight per kind (conflict kinds are summed).
    // Derived evidence added here: .phone (+6) if any bodyText/snippet contains one of input.phonesE164 or its last 7 digits;
    //   .company (+3) if group company ≈ input.company (jaroWinkler ≥ 0.9) and no company evidence exists yet;
    //   .conflict (−3, detail "Different company: X") if both companies known and jaroWinkler < 0.5;
    //   .username (+1.5) if two findings in the group share a username on different hosts;
    //   .avatar (+2) if two findings share the same avatarURL on different hosts.
    // score = sum(weights). status: ≥ auto → .auto, ≥ pending → .pending, else dropped.
    // Candidate.primaryURL = first non-linkedin finding url if any finding has emailHash evidence, else the linkedin url if present, else first url.
    // displayName/headline/company/location/avatarURL = first non-nil across findings in priority order gravatar, github, serp(linkedin), serp, page, username.
    // pages: one SourcePage per finding (kind = finding.pageKind, title = pageTitle, snippet, bodyText).
    // Sorted by score desc. If more than one candidate is .auto, all become .pending (ambiguous).
